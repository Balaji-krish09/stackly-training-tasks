-- 1. Table Creation: departments
USE college_db;

DROP TABLE IF EXISTS employees;
DROP TABLE IF EXISTS departments;

CREATE TABLE departments (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50)
);


-- 2. Table Creation: employees
CREATE TABLE employees (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(100),
    salary DECIMAL(10, 2),
    dept_id INT,
    FOREIGN KEY (dept_id) REFERENCES departments(dept_id)

-- 5. INNER JOIN Query
SELECT e.emp_name, d.dept_name
FROM employees e
INNER JOIN departments d ON e.dept_id = d.dept_id;


-- 6. LEFT OUTER JOIN Query
SELECT e.emp_name, d.dept_name
FROM employees e
LEFT OUTER JOIN departments d ON e.dept_id = d.dept_id;


-- 7. RIGHT OUTER JOIN Query
SELECT d.dept_name, e.emp_name
FROM employees e
RIGHT OUTER JOIN departments d ON e.dept_id = d.dept_id;


-- 8. Simulated FULL OUTER JOIN Query
SELECT e.emp_name, d.dept_name
FROM employees e
LEFT JOIN departments d ON e.dept_id = d.dept_id
UNION
SELECT e.emp_name, d.dept_name
FROM employees e
RIGHT JOIN departments d ON e.dept_id = d.dept_id;


-- 9. CROSS JOIN Query
SELECT e.emp_name, d.dept_name
FROM employees e
CROSS JOIN departments d;


-- 10. Total rows returned by CROSS JOIN Query
SELECT COUNT(*) AS total_rows
FROM employees e
CROSS JOIN departments d;

# 1. Create a list containing at least 10 elements
tech_stack = ["Python", "MySQL", "Git", "VS Code", "Java", "C++", "HTML", "CSS", "JavaScript", "Docker"]

# 5. Remove an element using remove()

# 6. Remove the last element using pop()
tech_stack.pop()
print(f"After pop(): {tech_stack}")

# 7. Remove an element at a specific index using pop(index)
tech_stack.pop(4)
print(f"After pop(index): {tech_stack}")

# 8. Count the occurrences of an element using count()

# 9. Find the index of an element using index()
