SET AUTOCOMMIT = 0;
START TRANSACTION;

-- Operation 1: Deduct ₹1000 from account 1001
UPDATE bank_accounts 
SET balance = balance - 1000.00 
WHERE account_id = 1001;

-- Operation 2: Add ₹1000 to account 1002
UPDATE bank_accounts 
SET balance = balance + 1000.00 
WHERE account_id = 1002;

-- Finalize persistent disk modifications
COMMIT;

-- 5. Display the table state after the COMMIT block execution

-- 6. Transaction Setup B: Balance modification statements with ROLLBACK
START TRANSACTION;

-- Operation 1: Attempt balance updates across account 1003
UPDATE bank_accounts 
SET balance = balance + 5000.00 
WHERE account_id = 1003;

-- Operation 2: Attempt balance updates across account 1004
UPDATE bank_accounts 
SET balance = balance - 3000.00 
WHERE account_id = 1004;

-- Discard uncommitted data buffer adjustments back to the previous stable baseline state
ROLLBACK;

-- 7. Display the table state after the ROLLBACK statement execution

SELECT * FROM bank_accounts;
SET AUTOCOMMIT = 1;


-- ==============================================================================
-- TECHNICAL REVIEW: ACID Properties in Relational Database Engines
-- ==============================================================================
/*
1. ATOMICITY:
   Ensures that a transaction is treated as a single unit of work. Either all operations 
   (Deduct and Credit) succeed completely, or the entire transaction fails, leaving 
   the database unchanged. Partial changes are never saved.

2. CONSISTENCY:
   Guarantees that a transaction transforms the database from one valid state to another, 
   maintaining all structural schemas, unique keys, check constraints, and referential integrity flags.

3. ISOLATION:
   Ensures that concurrent execution of transactions happens independently without cross-leakage. 
   Uncommitted modifications from one execution context remain completely invisible to 
   other concurrent queries.

4. DURABILITY:
   Guarantees that once a transaction commits successfully, its modifications are permanently written 
   to the non-volatile storage disk. The changes will not be lost even in the event of a total 
   system crash or power failure.
*/
